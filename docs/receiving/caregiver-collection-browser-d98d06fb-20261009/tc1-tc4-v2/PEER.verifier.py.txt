def identities(text):
    raw = text.encode('utf-8')
    return {'bytes': len(raw), 'sha256': hashlib.sha256(raw).hexdigest(),
            'git': hashlib.sha1(b'blob ' + str(len(raw)).encode('ascii') + b'\0' + raw).hexdigest()}

names = {'original-v4': 'originalV4', 'original-successor': 'originalDriver',
         'original-cases': 'originalCases', 'amended-driver': 'driver',
         'first-case-draft': 'firstCases', 'amended-cases': 'cases'}
digest_results = []
for declaration in proof['sourceDigests']:
    key = names[declaration['name']]
    observed = identities(s[key])
    for field in ['bytes', 'sha256', 'git']:
        assert observed[field] == declaration[field], (key, field)
    digest_results.append({'name': declaration['name'], **observed, 'exact': True})
extra_pins = {'proof': '85a00126ed5b739c79a3be927fd5dbe00038b2b7',

def exact_edits(old, new, edits, before_key='before', after_key='after', midpoint=None):
    current = old
    forward = []
    for index, edit in enumerate(edits):
        before, after = edit[before_key], edit[after_key]
        count = current.count(before)
        assert count == 1, ('forward', edit['label'], count)
        current = current.replace(before, after, 1)
        forward.append({'label': edit['label'], 'forwardOccurrences': count})
        if midpoint is not None and index == midpoint[0]:
            assert current == midpoint[1], ('first draft', index)
    assert current == new, 'complete forward body'
    for edit in reversed(edits):
        before, after = edit[before_key], edit[after_key]
        count = current.count(after)
        assert count == 1, ('inverse', edit['label'], count)
        current = current.replace(after, before, 1)
    assert current == old, 'complete inverse body'
    return {'edits': len(edits), 'forwardExact': True, 'inverseExact': True,
            'allForwardAndInverseOccurrencesOne': True, 'steps': forward}

def inventory(text):
    match = re.search(r'^const sourcePaths=(\[.*\]);$', text, re.M)
    assert match
    return json.loads(match.group(1))

syntax_commands = []
for section, expected_sources in [('initialParse', [s['driver'], s['firstCases']]),
                                  ('finalChangedCaseParse', [s['cases']])]:
    item = syntax[section]
    argv = shlex.split(item['exactCommand'])
    assert len(argv) == 5 and argv[1:3] == ['--experimental-vm-modules', '-e']
    assert argv[3] == syntax['parserSource']
    files = json.loads(argv[4])
    assert [x['text'] for x in files] == expected_sources
    assert item['actual']['exit_code'] == 0
    output = item['actual']['output']
    parsed_stdout, end = json.JSONDecoder().raw_decode(output)
    warning = output[end:]
    assert 'ExperimentalWarning: VM Modules' in warning
    assert parsed_stdout['linked'] is False and parsed_stdout['evaluated'] is False
    assert parsed_stdout['candidateBrowserServerOrIDBCalls'] == 0
    for file, observed in zip(files, parsed_stdout['files'], strict=True):
        assert file['path'] == observed['path']
        assert observed['status'] == 'unlinked'
        digest = identities(file['text'])
        for field in ['bytes', 'sha256', 'git']:
            assert observed[field] == digest[field]
    syntax_commands.append({'section': section, 'originalToolChunk': item['actual']['chunk_id'],
                            'exit': 0, 'exactCommandPayloadJoined': True,
                            'candidateEvaluated': False, 'reviewerReplay': False})
